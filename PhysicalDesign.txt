-- Student table for authentication and course identification
CREATE TABLE Student (
    StudentID INT AUTO_INCREMENT PRIMARY KEY,
    IDNumber VARCHAR(50) NOT NULL UNIQUE,
    FirstName VARCHAR(50) NOT NULL,
    LastName VARCHAR(50) NOT NULL,
    Email VARCHAR(100) UNIQUE NOT NULL,
    Course VARCHAR(50),
    YearLevel INT,
    Password VARCHAR(255) NOT NULL,
    CreatedAt DATETIME DEFAULT CURRENT_TIMESTAMP
); 

  

-- Teacher table for evaluation targets
CREATE TABLE Teacher (
    TeacherID INT AUTO_INCREMENT PRIMARY KEY,
    FirstName VARCHAR(50) NOT NULL,
    LastName VARCHAR(50) NOT NULL,
    Email VARCHAR(100) UNIQUE,
    Department VARCHAR(50),
    CreatedAt DATETIME DEFAULT CURRENT_TIMESTAMP
); 

  

-- Admin table for form creators
CREATE TABLE Admin (
    AdminID INT AUTO_INCREMENT PRIMARY KEY,
    FirstName VARCHAR(50) NOT NULL,
    LastName VARCHAR(50) NOT NULL,
    Email VARCHAR(100) UNIQUE NOT NULL,
    AccessLevel VARCHAR(50),
    Password VARCHAR(255) NOT NULL,
    CreatedAt DATETIME DEFAULT CURRENT_TIMESTAMP
); 

  

-- Subject/Course assignments (links teachers to courses they teach)
CREATE TABLE Subject (
    SubjectID INT AUTO_INCREMENT PRIMARY KEY,
    SubjectName VARCHAR(100) NOT NULL,
    TeacherID INT,
    Course VARCHAR(50),
    Section VARCHAR(20),
    FOREIGN KEY (TeacherID) REFERENCES Teacher(TeacherID) ON DELETE SET NULL
); 

  

-- Evaluation Forms (created by admins for students to complete)
CREATE TABLE EvaluationForm (
    EvaluationID INT AUTO_INCREMENT PRIMARY KEY,
    Title VARCHAR(100) NOT NULL,
    Description TEXT,
    CreatedBy INT,
    TeacherID INT,
    TargetCourse VARCHAR(50) DEFAULT 'All',
    DueDate DATE,
    IsActive BOOLEAN DEFAULT TRUE,
    DateCreated DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (CreatedBy) REFERENCES Admin(AdminID),
    FOREIGN KEY (TeacherID) REFERENCES Teacher(TeacherID)
); 

 

-- Comments/Feedback (optional feature)
CREATE TABLE Comment (
    CommentID INT AUTO_INCREMENT PRIMARY KEY,
    StudentID INT,
    TeacherID INT,
    Content TEXT,
    DateSubmitted DATETIME DEFAULT CURRENT_TIMESTAMP,
    Status VARCHAR(20) DEFAULT 'pending',
    FOREIGN KEY (StudentID) REFERENCES Student(StudentID),
    FOREIGN KEY (TeacherID) REFERENCES Teacher(TeacherID)
); 

  

-- Reports/Summaries generated for teachers
CREATE TABLE Report (
    ReportID INT AUTO_INCREMENT PRIMARY KEY,
    TeacherID INT,
    EvaluationID INT,
    SubmissionDate DATETIME DEFAULT CURRENT_TIMESTAMP,
    AverageScore DECIMAL(5,2),
    ResponseCount INT,
    ReportData JSON,
    FOREIGN KEY (TeacherID) REFERENCES Teacher(TeacherID),
    FOREIGN KEY (EvaluationID) REFERENCES EvaluationForm(EvaluationID)
); 

  

-- Survey Questions within each evaluation form
CREATE TABLE SurveyQuestion (
    QuestionID INT AUTO_INCREMENT PRIMARY KEY,
    EvaluationID INT,
    QuestionText TEXT NOT NULL,
    Category VARCHAR(50),
    QuestionType ENUM('rating', 'text', 'yesno', 'multiplechoice') DEFAULT 'rating',
    OrderIndex INT DEFAULT 0,
    IsRequired BOOLEAN DEFAULT TRUE,
    MinRating INT,
    MaxRating INT,
    FOREIGN KEY (EvaluationID) REFERENCES EvaluationForm(EvaluationID) ON DELETE CASCADE
);

-- Options for Multiple Choice questions
CREATE TABLE QuestionOption (
    OptionID INT AUTO_INCREMENT PRIMARY KEY,
    QuestionID INT,
    OptionText VARCHAR(255) NOT NULL,
    OrderIndex INT DEFAULT 0,
    FOREIGN KEY (QuestionID) REFERENCES SurveyQuestion(QuestionID) ON DELETE CASCADE
);

-- Form Submissions (tracks which student completed which form)
CREATE TABLE FormSubmission (
    SubmissionID INT AUTO_INCREMENT PRIMARY KEY,
    StudentID INT,
    EvaluationID INT,
    SubmittedAt DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (StudentID) REFERENCES Student(StudentID),
    FOREIGN KEY (EvaluationID) REFERENCES EvaluationForm(EvaluationID),
    UNIQUE KEY unique_submission (StudentID, EvaluationID)
); 

  

-- Individual question responses (linked to submission, not directly to student)
CREATE TABLE SurveyResponse (
    ResponseID INT AUTO_INCREMENT PRIMARY KEY,
    SubmissionID INT,
    QuestionID INT,
    Answer TEXT,
    FOREIGN KEY (SubmissionID) REFERENCES FormSubmission(SubmissionID) ON DELETE CASCADE,
    FOREIGN KEY (QuestionID) REFERENCES SurveyQuestion(QuestionID)
);
